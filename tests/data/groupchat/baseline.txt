# Normal operation: an introducer catching up before it introduces, explicit
# numbering in control messages, acks resolved by every reader, and backfill
# keeping a stream readable for a member who returns within the Sent-box
# window.

new-group alice bob carol
introduce carol dave
----
alice@0 not found
bob@0 not found
carol@0 introduces dave numbers 3:dave

read alice carol
----
carol@0 introduces dave numbers 3:dave
carol@1 not found

send alice hi            # a control message numbering dave goes first
----
alice@0 numbers 3:dave acks 0:carol@0
alice@1 "hi"

roster alice alice
----
carol alice bob dave

read bob carol
read bob alice
----
alice@0 numbers 3:dave acks 0:carol@0
alice@1 "hi"
alice@2 not found

roster bob alice
----
carol alice bob dave

send dave hello
read alice dave
send alice again         # dave is alice's index 3
----
alice@2 "again" acks 3:dave@1

read bob dave
read bob alice
acks bob alice 2
----
dave=dave@1

read carol alice
acked carol
----
alice:carol@0

# Bob is away for three epochs; alice's backfill keeps her boxes alive.
send alice while-bob-away
tick 3
read bob alice
----
alice@3 "while-bob-away"
alice@4 not found
